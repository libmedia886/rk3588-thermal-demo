cmake_minimum_required(VERSION 3.16)
project(rk3588_thermal_demo C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(LIBMEDIA_LIBRARY "" CACHE FILEPATH "Path to an installed libmedia.so or libmedia.a")
if(NOT LIBMEDIA_LIBRARY)
    find_library(LIBMEDIA_LIBRARY NAMES media libmedia HINTS ENV LIBMEDIA_ROOT PATH_SUFFIXES lib lib64 build src)
endif()
if(NOT LIBMEDIA_LIBRARY)
    message(FATAL_ERROR "Pass -DLIBMEDIA_LIBRARY=/absolute/path/to/libmedia.so or set LIBMEDIA_ROOT")
endif()
add_executable(thermal_palette_demo src/thermal_palette_demo.c)
target_include_directories(thermal_palette_demo PRIVATE include)
target_link_libraries(thermal_palette_demo PRIVATE ${LIBMEDIA_LIBRARY} m pthread dl)
